import numpy as np import pandas as pd import statsmodels.api as sm from statsmodels.tsa.stattools import adfuller # 1. 資料讀取與清洗 (Data Cleaning) # 假設從 AREMOS 匯出的 CSV 包含欄位: Date, MRT_Ridership, YouBike_Ridership, Gas_Price, Precipitation df = pd.read_csv("aremos_data.csv") # 轉換時間格式並設為 Index df["Date"] = pd.to_datetime(df["Date"]) df.set_index("Date", inplace=True) # 處理缺失值 (線性插補) 與取對數 (Log Transformation 以取得價格彈性) df = df.interpolate(method="linear") df["ln_mrt"] = np.log(df["MRT_Ridership"]) df["ln_youbike"] = np.log(df["YouBike_Ridership"]) df["ln_gas"] = np.log(df["Gas_Price"]) # 2. 單根檢定 (Augmented Dickey-Fuller Test) - 驗證時間序列平穩性 def check_stationarity(timeseries, name): result = adfuller(timeseries.dropna()) print(f"--- ADF Test for {name} ---") print(f"ADF Statistic: {result[0]:.4f}") print(f"p-value: {result[1]:.4f}") if result[1] <= 0.05: print("結論: 序列為平穩序列 (Reject H0)\n") else: print("結論: 序列存在單根，需進行一階差分 (Fail to reject H0)\n") check_stationarity(df["ln_mrt"], "Log(捷運運量)") check_stationarity(df["ln_gas"], "Log(油價)") # 若序列不平穩，建立一階差分欄位 df["d_ln_mrt"] = df["ln_mrt"].diff() df["d_ln_youbike"] = df["ln_youbike"].diff() df["d_ln_gas"] = df["ln_gas"].diff() # 3. 迴歸模型建構 (OLS Regression - 估算交叉價格彈性) # 模型：d_ln_mrt = β0 + β1 * d_ln_gas + β2 * Precipitation + ε X = df[["d_ln_gas", "Precipitation"]].dropna() Y = df["d_ln_mrt"].loc[X.index] # 加入常數項 (Intercept) X = sm.add_constant(X) # 估算 OLS 迴歸（採用 Robust Standard Errors 修正異質變異與自相關） model = sm.OLS(Y, X).fit(cov_type="HAC", cov_kwds={"maxlags": 3}) # 輸出統計結果報告 print(model.summary())
